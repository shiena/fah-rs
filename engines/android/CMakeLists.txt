cmake_minimum_required(VERSION 3.22)
project(fah_engine_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only arm64-v8a is supported")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ncnn_DIR "${ROOT}/third_party/ncnn-20261020-android-vulkan/${ANDROID_ABI}/lib/cmake/ncnn")
find_package(ncnn REQUIRED)
get_target_property(ncnn_options ncnn INTERFACE_COMPILE_OPTIONS)
list(REMOVE_ITEM ncnn_options "-fno-exceptions")
set_target_properties(ncnn PROPERTIES INTERFACE_COMPILE_OPTIONS "${ncnn_options}")
set(CPP "${ROOT}/backends/ncnn/cpp")
add_library(fah_backend SHARED "${CPP}/bridge.cpp" "${CPP}/runtime.cpp" "${CPP}/vulkan_ops.cpp")
target_compile_options(fah_backend PRIVATE -fexceptions -O3)
target_link_libraries(fah_backend PRIVATE ncnn android log dl)
