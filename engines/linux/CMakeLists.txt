cmake_minimum_required(VERSION 3.22)
project(fah_engine_linux LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
# Keep CPU kernels portable; ncnn selects ISA variants at runtime.
set(NCNN_OPENMP ON CACHE BOOL "" FORCE)
set(NCNN_RUNTIME_CPU ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/ncnn-src" ncnn)
set(CPP "${ROOT}/backends/ncnn/cpp")
add_library(fah_backend SHARED "${CPP}/bridge.cpp" "${CPP}/runtime.cpp" "${CPP}/vulkan_ops.cpp")
target_compile_options(fah_backend PRIVATE -fexceptions -O3)
target_link_libraries(fah_backend PRIVATE ncnn)
set_target_properties(fah_backend PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE)
