cmake_minimum_required(VERSION 3.22)
project(fah_windows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(NCNN_STDIO ON CACHE BOOL "" FORCE)
set(NCNN_STRING ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/ncnn-src" ncnn)
set(FAH_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../backends/ncnn/cpp")
add_library(fah_backend SHARED "${FAH_CPP}/bridge.cpp" "${FAH_CPP}/runtime.cpp" "${FAH_CPP}/vulkan_ops.cpp")
target_compile_options(fah_backend PRIVATE /EHsc /utf-8 /O2)
target_link_libraries(fah_backend PRIVATE ncnn)
